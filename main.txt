/*
  Ancient Rome Exhibit — ESP32-WROOM + PN532 (I2C)
  The Windows laptop runs the EXISTING Flask exhibit:
    Exhibition: http://192.168.10.100:5000
    Artifact editor: http://192.168.10.100:5001

  Install Arduino library: Seeed Arduino NFC
  PN532 mode switches/jumpers must be set to I2C.
*/

#include <WiFi.h>
#include <HTTPClient.h>
#include <Wire.h>
#include <PN532_I2C.h>
#include <PN532.h>

const char* WIFI_SSID = "TambanisDevNetwork";
const char* WIFI_PASSWORD = "PUT_YOUR_WIFI_PASSWORD_HERE";

// Laptop IPv4 address supplied by you:
const char* SERVER_BASE_URL = "http://192.168.10.100:5000";

// Must match config.json api_token on the laptop.
const char* API_TOKEN = "admin";

#define I2C_SDA 21
#define I2C_SCL 22

PN532_I2C pn532Interface(Wire);
PN532 nfc(pn532Interface);

String lastUID = "";
unsigned long lastPostMs = 0;
unsigned long lastHeartbeatMs = 0;
const unsigned long SCAN_REFRESH_MS = 1000;
const unsigned long HEARTBEAT_MS = 5000;

String uidToHex(const uint8_t* uid, uint8_t length) {
  String value;
  for (uint8_t i = 0; i < length; i++) {
    if (uid[i] < 0x10) value += "0";
    value += String(uid[i], HEX);
  }
  value.toUpperCase();
  return value;
}

bool postJson(const String& endpoint, const String& body) {
  if (WiFi.status() != WL_CONNECTED) return false;

  HTTPClient http;
  String url = String(SERVER_BASE_URL) + endpoint;
  http.begin(url);
  http.addHeader("Content-Type", "application/json");
  http.addHeader("X-Exhibit-Token", API_TOKEN);
  http.setConnectTimeout(2500);
  http.setTimeout(2500);

  int code = http.POST(body);
  String response = http.getString();

  Serial.print("POST ");
  Serial.print(endpoint);
  Serial.print(" -> ");
  Serial.println(code);
  if (response.length()) Serial.println(response);

  http.end();
  return code >= 200 && code < 300;
}

void sendHeartbeat() {
  if (WiFi.status() != WL_CONNECTED) return;
  String ip = WiFi.localIP().toString();
  String body = "{\"ip\":\"" + ip + "\"}";
  postJson("/api/reader", body);
  lastHeartbeatMs = millis();
}

void connectWiFi() {
  if (WiFi.status() == WL_CONNECTED) return;

  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  Serial.print("Connecting to Wi-Fi");

  unsigned long started = millis();
  while (WiFi.status() != WL_CONNECTED && millis() - started < 15000) {
    delay(400);
    Serial.print(".");
  }
  Serial.println();

  if (WiFi.status() == WL_CONNECTED) {
    Serial.print("Connected. ESP32 IP: ");
    Serial.println(WiFi.localIP());
  } else {
    Serial.println("Wi-Fi connection failed; will retry.");
  }
}

void setup() {
  Serial.begin(115200);
  delay(500);

  Wire.begin(I2C_SDA, I2C_SCL);
  connectWiFi();

  Serial.println("Starting PN532...");
  nfc.begin();
  uint32_t version = nfc.getFirmwareVersion();

  if (!version) {
    Serial.println("ERROR: PN532 not found. Check power, I2C mode and wiring.");
    while (true) delay(1000);
  }

  Serial.print("PN532 firmware ");
  Serial.print((version >> 16) & 0xFF);
  Serial.print(".");
  Serial.println((version >> 8) & 0xFF);

  nfc.SAMConfig();
  Serial.println("Ready. Scan an artifact tag.");
}

void loop() {
  if (WiFi.status() != WL_CONNECTED) {
    connectWiFi();
  }

  if (WiFi.status() == WL_CONNECTED &&
      millis() - lastHeartbeatMs >= HEARTBEAT_MS) {
    sendHeartbeat();
  }

  uint8_t uid[7];
  uint8_t uidLength = 0;
  bool found = nfc.readPassiveTargetID(
    PN532_MIFARE_ISO14443A, uid, &uidLength, 120
  );

  if (found) {
    String uidString = uidToHex(uid, uidLength);
    unsigned long now = millis();

    // Refresh about once a second while a tag remains on the stand.
    // The existing server uses these refreshes to know the artifact is still present.
    if (uidString != lastUID || now - lastPostMs >= SCAN_REFRESH_MS) {
      Serial.print("TAG ");
      Serial.println(uidString);

      String body = "{\"uid\":\"" + uidString + "\",\"source\":\"esp32\"}";
      if (postJson("/api/nfc", body)) {
        lastUID = uidString;
        lastPostMs = now;
      }
    }
    delay(40);
  } else {
    delay(40);
  }
}
