

#include <Wire.h>
#include <Adafruit_PN532.h>
#include <ArduinoJson.h> // Still used to build perfect JSON formatting

#define I2C_SDA 4
#define I2C_SCL 5
Adafruit_PN532 nfc(I2C_SDA, I2C_SCL);

void setup(void) {
  Serial.begin(115200);
  while (!Serial && millis() < 3000) delay(10);

  // Initialize PN532
  nfc.begin();
  uint32_t versiondata = nfc.getFirmwareVersion();
  if (!versiondata) {
    Serial.print("Didn't find PN532 board. Check wiring!");
    while (1);
  }
  nfc.setPassiveActivationRetries(0xFF);
  nfc.SAMConfig();
  
  Serial.println("--- NFC JSON Logger Ready ---");
}

void loop(void) {
  boolean success;
  uint8_t uid[] = { 0, 0, 0, 0, 0, 0, 0 };  
  uint8_t uidLength;                        
  
  success = nfc.readPassiveTargetID(PN532_MIFARE_ISO14443A, &uid, &uidLength);
  
  if (success) {
    // 1. Format the UID bytes to a clean Hex string
    String uidStr = "";
    for (uint8_t i = 0; i < uidLength; i++) {
      if (uid[i] < 0x10) uidStr += "0";
      uidStr += String(uid[i], HEX);
      if (i < uidLength - 1) uidStr += ":";
    }
    uidStr.toUpperCase();

    // 2. Create the JSON entry in memory
    JsonDocument doc;
    doc["timestamp_ms"] = millis();
    doc["uid"] = uidStr;

    // 3. Print the JSON string directly to the computer
    serializeJson(doc, Serial);
    Serial.println(); // Newline to separate entries
    
    delay(2000); // 2-second cooldown
  }
}
