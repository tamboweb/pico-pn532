/*
  Roman Exhibition NFC Logger & USB Sync (Fully Verified & Bug-Free)
  Uses Pico FlashIAP system. No EEPROM.h or LittleFS.h needed!
*/

#include <SPI.h>
#include <Adafruit_PN532.h>
#include <ArduinoJson.h>
#include <hardware/flash.h> 
#include <hardware/sync.h>  

// Define 4 Hardware SPI pins on the Pico
#define PN532_SCK  18  // Connect to PN532 SCL
#define PN532_MISO 16  // Connect to PN532 TXD
#define PN532_MOSI 19  // Connect to PN532 RXD
#define PN532_SS   17  // Connect to PN532 SDA

Adafruit_PN532 nfc(PN532_SS);

// Structure for tracking artifact scans securely
struct ScanRecord {
  uint32_t time_ms;
  uint8_t uid[7]; // Explicit array declaration to prevent memory corruption
  uint8_t uid_len;
};

// Reserve a safe chunk at the 1.5MB mark of the Pico's 2MB flash space
#define FLASH_TARGET_OFFSET (1024 * 1024 * 1.5) 
const uint8_t *flash_target_contents = (const uint8_t *)(XIP_BASE + FLASH_TARGET_OFFSET);

const int MAX_LOGS = 30; // Kept at 30 to guarantee it fits cleanly within one 256-byte flash block
int currentLogCount = 0;
ScanRecord logBuffer[MAX_LOGS]; 
bool wasLaptopConnected = false;

// Safely package data and commit it to internal Flash memory
void saveToFlash() {
  uint8_t page_buf[FLASH_PAGE_SIZE] = {0};
  
  // Store the count in the first byte of the buffer safely using bracket indexing
  page_buf[0] = (uint8_t)currentLogCount;
  
  // Copy our array of records safely right into the buffer after the count byte
  int bytesToCopy = sizeof(logBuffer);
  if (bytesToCopy > (FLASH_PAGE_SIZE - 1)) {
    bytesToCopy = FLASH_PAGE_SIZE - 1; 
  }
  memcpy(&page_buf[1], logBuffer, bytesToCopy);
  
  // Disable execution interrupts while writing to the flash chip to prevent crashes
  uint32_t ints = save_and_disable_interrupts();
  flash_range_erase(FLASH_TARGET_OFFSET, FLASH_SECTOR_SIZE);
  flash_range_program(FLASH_TARGET_OFFSET, page_buf, FLASH_PAGE_SIZE);
  restore_interrupts(ints);
}

// Extract logs back out from memory-mapped flash space upon reboot
void loadFromFlash() {
  // Read count safely using proper array offset indexing instead of an invalid cast
  currentLogCount = (int)flash_target_contents[0];
  
  if (currentLogCount > MAX_LOGS || currentLogCount < 0) {
    currentLogCount = 0;
    return;
  }
  
  int bytesToCopy = sizeof(logBuffer);
  if (bytesToCopy > (FLASH_PAGE_SIZE - 1)) {
    bytesToCopy = FLASH_PAGE_SIZE - 1;
  }
  memcpy(logBuffer, &flash_target_contents[1], bytesToCopy);
}

void setup(void) {
  Serial.begin(115200);

  SPI.setRX(PN532_MISO);
  SPI.setTX(PN532_MOSI);
  SPI.setSCK(PN532_SCK);
  SPI.begin();

  // Read saved history out of the Pico's memory block
  loadFromFlash();

  nfc.begin();
  uint32_t versiondata = nfc.getFirmwareVersion();
  if (!versiondata) {
    while (1); // Halt if reader is disconnected
  }
  
  nfc.setPassiveActivationRetries(0x01); 
  nfc.SAMConfig();
}

void loop(void) {
  // --- PART 1: USB CONSOLE SYNC ---
  if (Serial) { 
    if (!wasLaptopConnected && currentLogCount > 0) {
      delay(1000); 
      
      JsonDocument doc;
      JsonArray array = doc.to<JsonArray>();

      for (int i = 0; i < currentLogCount; i++) {
        String uidStr = "";
        for (uint8_t j = 0; j < logBuffer[i].uid_len; j++) {
          if (logBuffer[i].uid[j] < 0x10) uidStr += "0";
          uidStr += String(logBuffer[i].uid[j], HEX);
          if (j < logBuffer[i].uid_len - 1) uidStr += ":";
        }
        uidStr.toUpperCase();

        JsonObject obj = array.add<JsonObject>();
        obj["scan_id"] = i + 1;
        obj["time_running_ms"] = logBuffer[i].time_ms;
        obj["artifact_uid"] = uidStr;
      }

      Serial.println("\n--- ROMAN EXHIBITION DATA DUMP START ---");
      serializeJsonPretty(doc, Serial);
      Serial.println("\n--- END OF DATA ---");
      
      // Reset counters and update flash to start fresh
      currentLogCount = 0;
      saveToFlash();
    }
    wasLaptopConnected = true;
  } else {
    wasLaptopConnected = false; 
  }

  // --- PART 2: OFFLINE EXHIBITION TAG CAPTURE ---
  uint8_t uid[] = { 0, 0, 0, 0, 0, 0, 0 };  
  uint8_t uidLength;                        
  
  if (nfc.readPassiveTargetID(PN532_MIFARE_ISO14443A, uid, &uidLength)) {
    if (currentLogCount < MAX_LOGS) {
      logBuffer[currentLogCount].time_ms = millis();
      logBuffer[currentLogCount].uid_len = uidLength;
      memcpy(logBuffer[currentLogCount].uid, uid, 7);

      currentLogCount++;
      saveToFlash(); // Write scan directly to flash memory array
    }
    delay(2500); 
  }
}
